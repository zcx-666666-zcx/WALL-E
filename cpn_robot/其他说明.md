# 项目说明
# 功能说明
- 登录校验：校验用户名和密码
- 事件管理：查看所有事件，以列表形式在前端展示，并对其进行 增删改查 操作
- 联系人管理： 查看所有联系人，以列表形式在前端展示，并对其进行 增删查 操作
- 设备管理： 查看所有设备，以列表形式在前端展示，并对其进行 增删查 操作
- 命令查看： 查询本用户拥有的所有设备，并创建对应的子菜单，进入对应的子菜单查看当前设备执行过的命令（由于硬件为完善，暂无任何命令，可以在数据库中添加来展示）

# 表结构
create table user(
    id int not null primary key auto_increment comment "用户ID，唯一且自增",
    name varchar(10) not null comment "用户名",
    phone_number varchar(11) not null comment "手机号",
    password varchar(32) not null comment "密码"
)comment "用户表";

create table event(
    id int not null primary key auto_increment comment "事件ID，唯一且自增",
    event_name varchar(20) not null comment "事件名称",
    event_date date not null comment "事件日期",
    event_time datetime not null comment "事件时间",
    repeat_days int not null comment "重复天数",
    priority enum("low","medium","high") not null comment "优先级",
    machins varchar(20) not null comment "使用设备",
    description varchar(100) not null comment "描述",
    user_id int not null comment "用户ID"
) comment "事件表";

create table contract_user(
    id int not null primary key auto_increment comment "ID，唯一且自增",
    contract_name varchar(20) not null comment "被绑定人名称",
    contract_phone_number varchar(11) not null comment "被绑定人手机号"
    user_id int not null comment "用户ID"
) comment "联系人表";

create table contract_machine(
    id int not null primary key auto_increment comment "ID，唯一且自增",
    machines varchar(20) not null comment "被绑设备名称",
    connectionId varchar(10) not null comment "被绑定设备号",
user_id int not null comment "用户ID"
) comment "设备表";

create table command (
    id int not null primary key auto_increment comment "ID，唯一且自增",
    command_name varchar(100) comment '指令名称',
    command_code varchar(50) comment '指令代码',
    execution_status varchar(20) comment '执行状态',
    time datetime comment '执行时间',
    machines varchar(100) comment '设备名称',
    user_id int comment '用户ID',
) comment='指令执行记录表';
- **user_id为多表查询的唯一凭证**
